# Watch scope; called within the existing once-per-month observation.
nfp50_sample_supply = {
    set_variable = { name = nfp50_supply_critical value = 0 }
    # Exactly the vanilla goods-shortage red-alert semantics, including reader.
    if = { limit = { scope:nfp46_market_good = {
        market_goods_shortage_ratio > 0.33
        country_has_local_shortage = scope:nfp46_reader
        NOT = { goods = { this = g:merchant_marine } }
    } } set_variable = { name = nfp50_supply_critical value = 1 } }
    # Migration initializes only new observation bookkeeping, never old facts.
    if = { limit = { NOT = { has_variable = nfp50_supply_mark_episode } }
        set_variable = { name = nfp50_supply_mark_episode value = var:nfp46_episode }
        set_variable = { name = nfp50_supply_last_stage value = 0 }
        if = { limit = { var:nfp46_bad >= 3 } set_variable = { name = nfp50_supply_last_stage value = 1 } }
        if = { limit = { var:nfp46_bad >= 6 } set_variable = { name = nfp50_supply_last_stage value = 2 } }
        set_variable = { name = nfp50_supply_last_critical value = var:nfp50_supply_critical }
        set_variable = { name = nfp50_supply_last_period value = nfp20_period }
    }
}

# Candidate scope, only current problem7 facts emitted from the market observer.

    if = { limit = { var:nfp20_problem = 7 }

        set_variable = { name = nfp50_supply_period value = var:nfp20_period }

            set_variable = { name = nfp50_supply_followup value = 1 }
        }

# Candidate scope. Publication state is taken from the actual printed snapshot.
nfp50_mark_supply = {
    if = { limit = { has_variable = nfp50_supply exists = var:nfp46_watch var:nfp20_type = 3 }
        save_scope_as = nfp50_supply_printed
        var:nfp46_watch = {
            set_variable = { name = nfp50_supply_mark_episode value = scope:nfp50_supply_printed.var:nfp20_episode }
            set_variable = { name = nfp50_supply_last_stage value = scope:nfp50_supply_printed.var:nfp50_supply_stage }
            set_variable = { name = nfp50_supply_last_critical value = scope:nfp50_supply_printed.var:nfp50_supply_critical }
            # Cooldown starts when delivered, not at the potentially older event.
            set_variable = { name = nfp50_supply_last_period value = nfp20_period }
        }
    }
}

# Watch scope. Bounded followups add ordinary candidates through the old pipeline.
nfp50_supply_followup = {
    if = { limit = { NOT = { has_variable = nfp50_supply_follow_seen } }
        set_variable = { name = nfp50_supply_follow_seen value = -100000 }
    }
    if = { limit = { NOT = { var:nfp50_supply_follow_seen = nfp20_period } }
        set_variable = { name = nfp50_supply_follow_seen value = nfp20_period }
        if = { limit = { var:nfp46_active = 1 var:nfp46_bad >= 2 var:nfp46_published = var:nfp46_episode }
            # An old pre-update published episode gets a conservative baseline.
            if = { limit = { NOT = { var:nfp50_supply_mark_episode = var:nfp46_episode } }
                set_variable = { name = nfp50_supply_mark_episode value = var:nfp46_episode }
                set_variable = { name = nfp50_supply_last_stage value = 0 }
                set_variable = { name = nfp50_supply_last_critical value = var:nfp50_supply_critical }
                set_variable = { name = nfp50_supply_last_period value = nfp20_period }
            }
            if = { limit = { OR = {
                AND = { var:nfp50_supply_critical = 1 var:nfp50_supply_last_critical = 0 }
                AND = { nfp50_supply_pub_gap >= 3 OR = {
                    AND = { var:nfp46_bad >= 3 var:nfp50_supply_last_stage < 1 }
                    AND = { var:nfp46_bad >= 6 var:nfp50_supply_last_stage < 2 }
                } }
            } }
                save_scope_as = nfp46_watch
                scope:nfp46_reader = {
                    if = { limit = { NOT = { any_in_list = { variable = nfp20_candidates
                        var:nfp20_used = 0 nfp20_candidate_age <= 12 var:nfp20_type = 3 exists = var:nfp46_watch
                        var:nfp46_watch = scope:nfp46_watch var:nfp20_episode = scope:nfp46_watch.var:nfp46_episode
                    } } }
                        nfp50_emit_supply_followup = yes
                    }
                }
            }
        }
    }
}

# Reader scope. Restore the previous printed warning mode until this fact is
# actually selected. A pruned pending followup cannot change recovery wording.
nfp50_emit_supply_followup = {
    scope:nfp46_watch = {
        set_variable = { name = nfp50_supply_previous_mode value = var:nfp46_mode }
        set_variable = { name = nfp46_mode value = 2 }
        if = { limit = { var:nfp46_native = 1 } set_variable = { name = nfp46_mode value = 1 } }
    }
    nfp46_emit_warning = yes
    scope:nfp46_watch = {
        set_variable = { name = nfp46_mode value = var:nfp50_supply_previous_mode }
        remove_variable = nfp50_supply_previous_mode
    }
}
