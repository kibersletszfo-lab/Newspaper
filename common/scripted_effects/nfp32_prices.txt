# Reader-owned bounded cache, updated once per calendar month (or market switch).
# No candidates, fictional discoveries, world scans, or balance effects.
nfp32_observe_prices = {
    save_scope_as = nfp32_owner
    if = { limit = { exists = capital.market }
        if = { limit = { OR = {
            NOT = { has_variable = nfp32_period }
            NOT = { var:nfp32_period = nfp20_period }
            NOT = { exists = var:nfp32_market }
            NOT = { var:nfp32_market = capital.market }
        } }
            if = { limit = { OR = { NOT = { exists = var:nfp32_market } NOT = { var:nfp32_market = capital.market } } }
                every_in_list = { variable = nfp32_prices destroy_container = yes }
                clear_variable_list = nfp32_prices
                set_variable = { name = nfp32_count value = 0 }
            }
            set_variable = { name = nfp32_market value = capital.market }
            set_variable = { name = nfp32_market_owner value = capital.market.owner.country_definition }
            capital.market = {
                every_market_goods = {
                    limit = { OR = { market_goods_buy_orders > 0 market_goods_sell_orders > 0 } }
                    save_scope_as = nfp32_market_good
                    goods = { save_scope_as = nfp32_good }
                    scope:nfp32_owner = {
                        if = { limit = { NOT = { any_in_list = { variable = nfp32_prices var:nfp32_good = scope:nfp32_good } } var:nfp32_count < 128 }
                            change_variable = { name = nfp32_count add = 1 }
                            create_container = { tags = { nfp32_price } parent = scope:nfp32_owner save_scope_as = nfp32_entry }
                            scope:nfp32_entry = {
                                set_variable = { name = nfp32_good value = scope:nfp32_good }
                                set_variable = { name = nfp32_index value = scope:nfp32_owner.var:nfp32_count }
                            }
                            add_to_variable_list = { name = nfp32_prices target = scope:nfp32_entry }
                        }
                        every_in_list = { variable = nfp32_prices limit = { var:nfp32_good = scope:nfp32_good }
                            set_variable = { name = nfp32_comparable value = 0 }
                            if = { limit = { has_variable = nfp32_period nfp32_gap = 1 var:nfp32_current > 0 }
                                set_variable = { name = nfp32_previous value = var:nfp32_current }
                                set_variable = { name = nfp32_comparable value = 1 }
                            }
                            set_variable = { name = nfp32_current value = scope:nfp32_market_good.nfp32_price }
                            set_variable = { name = nfp32_period value = nfp20_period }
                        }
                    }
                }
            }
            set_variable = { name = nfp32_period value = nfp20_period }
        }
    } else = {
        # Invalidate comparisons across missing-market intervals, even within month.
        every_in_list = { variable = nfp32_prices destroy_container = yes }
        clear_variable_list = nfp32_prices
        remove_variable = nfp32_market
        remove_variable = nfp32_market_owner
        remove_variable = nfp32_period
        set_variable = { name = nfp32_count value = 0 }
    }
}
