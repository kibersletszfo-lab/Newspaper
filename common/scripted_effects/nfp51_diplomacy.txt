# Reader root. Named countries were captured before changing scopes.
nfp51_emit = {
 save_scope_as = nfp51_reader
 # The closing account replaces ONLY unpublished opening/join dispatches of
 # this same crisis. Their old pages in published issues remain untouched.
 if = { limit = { scope:nfp51_play = { var:nfp51_stage = 30 } }
  every_in_list = { variable = nfp20_candidates
   limit = { has_variable = nfp51_schema var:nfp20_used = 0
    var:nfp51_crisis = scope:nfp51_play.var:nfp51_crisis
    OR = { var:nfp20_type = 29 var:nfp20_type = 31 }
   }
   set_variable = { name = nfp20_used value = 1 }
  }
 }
 if = { limit = { NOT = { has_variable = nfp20_source_serial } }
  set_variable = { name = nfp20_source_serial value = 0 }
 }
 change_variable = { name = nfp20_candidate_seq add = 1 }
 change_variable = { name = nfp20_source_serial add = 1 }
 create_container = { tags = { nfp20_candidate } parent = scope:nfp51_reader save_scope_as = nfp51_candidate }

  set_variable = { name = nfp20_used value = 0 }
  set_variable = { name = nfp20_source_definition value = scope:nfp51_initiator.country_definition }
  set_variable = { name = nfp20_secondary value = scope:nfp51_target.country_definition }
  set_variable = { name = nfp20_period value = nfp20_period }
  set_variable = { name = nfp20_year value = year }
  set_variable = { name = nfp20_month value = month }
  set_variable = { name = nfp20_before value = 0 }
  set_variable = { name = nfp20_after value = 1 }

# Play root. Classify actual relations at event time, without scanning populations.
nfp51_distribute = {
 scope:nfp51_initiator = { save_scope_as = nfp63_source }
 scope:nfp51_target = { save_scope_as = nfp63_second }
 every_country = {
  limit = { is_player = yes }
  nfp20_migrate = yes
  nfp63_kind_set = { kind = $type$ }
  nfp63_assess = yes
  set_variable = { name = nfp51_relation value = 0 }
  if = { limit = { OR = { this = scope:nfp51_initiator this = scope:nfp51_target } }
   set_variable = { name = nfp51_relation value = 1 }
  } else_if = { limit = { OR = {
   any_scope_treaty = { binds = scope:nfp51_initiator any_scope_article = { has_type = alliance } }
   any_scope_treaty = { binds = scope:nfp51_target any_scope_article = { has_type = alliance } }
  } } set_variable = { name = nfp51_relation value = 2 }
  } else_if = { limit = { OR = { is_in_same_power_bloc = scope:nfp51_initiator is_in_same_power_bloc = scope:nfp51_target } }
   set_variable = { name = nfp51_relation value = 3 }
  } else_if = { limit = { scope:nfp51_play = { var:nfp51_stage = 31 } this = scope:nfp51_actor }
   set_variable = { name = nfp51_relation value = 4 }
  } else_if = { limit = { scope:nfp51_play = { var:nfp51_stage = 31 }
   any_scope_treaty = { binds = scope:nfp51_actor any_scope_article = { has_type = alliance } }
  } set_variable = { name = nfp51_relation value = 5 }
  } else_if = { limit = { scope:nfp51_play = { var:nfp51_stage = 31 } is_in_same_power_bloc = scope:nfp51_actor }
   set_variable = { name = nfp51_relation value = 6 }
  }
  if = { limit = { OR = { var:nfp51_relation > 0 var:nfp63_allow = 1 } }
   set_variable = { name = nfp63_allow value = 1 }
   if = { limit = { var:nfp63_reason = 0 } set_variable = { name = nfp63_reason value = 4 } }
   nfp51_emit = { type = $type$ stage = $stage$ priority = $priority$ }
  }
 }
}

# One play-owned record per joining country, at most 256 records.
# Unlike an actor-local last-ID marker, this survives interleaved crises.
# All records die with their parent play; no record is copied to an issue.
nfp51_record_join = {
 set_variable = { name = nfp51_should_emit value = 1 }
 every_in_list = { variable = nfp51_joiners
  limit = { var:nfp51_actor = { this = scope:nfp51_actor.country_definition } }
  if = { limit = { var:nfp51_side = { this = scope:nfp51_side.country_definition } }
   scope:nfp51_play = { set_variable = { name = nfp51_should_emit value = 0 } }
  } else = {
   set_variable = { name = nfp51_side value = scope:nfp51_side.country_definition }
  }
 }
 if = { limit = { var:nfp51_joiner_count < 256
  NOT = { any_in_list = { variable = nfp51_joiners
   var:nfp51_actor = { this = scope:nfp51_actor.country_definition }
  } }
 }
  create_container = { tags = { nfp51_joiner } parent = scope:nfp51_play save_scope_as = nfp51_join_record }
  scope:nfp51_join_record = {
   set_variable = { name = nfp51_actor value = scope:nfp51_actor.country_definition }
   set_variable = { name = nfp51_side value = scope:nfp51_side.country_definition }
  }
  add_to_variable_list = { name = nfp51_joiners target = scope:nfp51_join_record }
  change_variable = { name = nfp51_joiner_count add = 1 }
 }
}
