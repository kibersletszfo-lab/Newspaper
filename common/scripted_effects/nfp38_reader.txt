nfp38_refresh_unread = {
save_scope_as = nfp38_counter_owner
set_variable = { name = nfp38_unread_count value = 0 }
every_in_list = {
variable = nfp_issues
limit = { has_variable = nfp38_unread }
scope:nfp38_counter_owner = { change_variable = { name = nfp38_unread_count add = 1 } }
}
}
nfp38_mark_selected_read = {
if = { limit = { has_variable = nfp_selected_issue }
var:nfp_selected_issue = { remove_variable = nfp38_unread }
}
nfp38_refresh_unread = yes
}
nfp38_choose_unread = {
save_scope_as = nfp38_open_owner
ordered_in_list = {
variable = nfp_issues
limit = { has_variable = nfp38_unread }
order_by = nfp38_newest_order
position = 0
check_range_bounds = no
save_scope_as = nfp38_open_issue
scope:nfp38_open_owner = { set_variable = { name = nfp_selected_issue value = scope:nfp38_open_issue } }
}
}
nfp38_open_reader = {
 nfp78_prepare_reader = yes
nfp_initialize = yes
nfp_refresh_archive = yes
nfp38_choose_unread = yes
set_variable = { name = nfp_window_open value = 1 }
nfp38_mark_selected_read = yes
}
nfp38_close_reader = {
remove_variable = nfp_setup_pending
set_variable = { name = nfp_window_open value = 0 }
nfp51_finish_special = yes
}
nfp38_receive_issue = {
nfp38_assign_header = yes
scope:nfp_new_issue = {
if = { limit = { var:nfp_kind < 3 } set_variable = nfp38_unread }
set_variable = { name = nfp38_header_id value = scope:nfp_reader.var:nfp38_header_id }
set_variable = { name = nfp39_header_code value = scope:nfp_reader.var:nfp39_header_code }
}
