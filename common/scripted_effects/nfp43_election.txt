# Reader country. New generation invalidates old party baselines without scanning them.
nfp43_election_start = {
    if = { limit = { is_player = yes }
        if = { limit = { NOT = { has_variable = nfp43_campaign } }
            set_variable = { name = nfp43_campaign value = 0 }
        }
        change_variable = { name = nfp43_campaign add = 1 }
        nfp43_clear_election_candidates = yes
        remove_variable = nfp43_election_printed_today
    }
}

# Reader country; one pending bulletin at most. Called before refresh and campaign end.
nfp43_clear_election_candidates = {
    save_scope_as = nfp43_election_owner
    every_in_list = { variable = nfp20_candidates
        limit = { var:nfp20_type = 24 }
        save_scope_as = nfp43_election_remove
        scope:nfp43_election_owner = {
            remove_list_variable = { name = nfp20_candidates target = scope:nfp43_election_remove }
            change_variable = { name = nfp20_queue_count add = -1 }
        }
        destroy_container = yes
    }
}

        ordered_active_party = {

            if = { limit = { scope:nfp20_candidate = { var:nfp43_rows < 6 } }

                        set_variable = { name = nfp43_row_1 value = flag:NFP43_ELECTION_ROW_CURRENT_1 }
