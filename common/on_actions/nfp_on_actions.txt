nfp20_war_start = {
        trigger = { NOT = { nfp52_us_civil_war_play = yes }  NOT = { has_variable = nfp20_war_start_recorded } exists = initiator exists = target }
        effect = {
            set_variable = nfp20_war_start_recorded
            target = { save_scope_as = nfp20_other_actor }
            initiator = { if = { limit = { has_variable = nfp20_observed_period }
                nfp20_prepare = yes
                set_variable = { name = nfp20_f_secondary value = scope:nfp20_other_actor.country_definition }
                set_variable = { name = nfp20_f_secondary_country value = scope:nfp20_other_actor }
                set_variable = { name = nfp20_f_urgent value = 1 }
                if = { limit = { always = yes }
                    set_variable = { name = nfp20_f_before value = 0 }
set_variable = { name = nfp20_f_after value = 0 }
set_variable = { name = nfp20_f_extra value = 0 }
set_variable = { name = nfp20_f_detail value = flag:NFP20_NONE }
nfp20_emit = { type = 10 section = 2 body = NFP20_WAR_START severity = 95 global = yes }
                }
            } }
            every_country = { limit = { is_player = yes has_variable = nfp20_schema } nfp_maybe_urgent = yes }
        }
    }
nfp20_war_end = {
        trigger = { NOT = { has_variable = nfp20_war_end_recorded } exists = scope:actor exists = scope:target }
        effect = {
            set_variable = nfp20_war_end_recorded
            scope:target = { save_scope_as = nfp20_other_actor }
            scope:actor = { if = { limit = { has_variable = nfp20_observed_period }
                nfp20_prepare = yes
                set_variable = { name = nfp20_f_secondary value = scope:nfp20_other_actor.country_definition }
